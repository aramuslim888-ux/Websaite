import cv2 import numpy as np from ultralytics import YOLO # 1. LOAD MODEL & VIDEO model = YOLO("yolov8m.pt") video_path = "video.mp4" # ناوی ئەو ڤیدۆیەی کە دەتەوێت شیکاری بۆ بکەیت cap = cv2.VideoCapture(video_path) # زانیارییە سەرەتاییەکانی ڤیدۆ width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)) height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)) fps = cap.get(cv2.CAP_PROP_FPS) # ڕێکخستنی تۆمارکردنی ڤیدۆی دەرچوو (Output Video) out = cv2.VideoWriter('output_speed.mp4', cv2.VideoWriter_fourcc(*'mp4v'), fps, (width, height)) # دیاریکردنی هێڵەکان یان ناوچەکان بۆ پێوانەکردنی خێرایی # دەتوانیت بەپێی قەبارەی ڤیدۆکەت ئەم خاڵانە بگۆڕیت pts = np.array([[100, 500], [400, 500], [500, 700], [200, 700]], np.int32) while cap.isOpened(): success, frame = cap.read() if not success: break # بەکارهێنانی مۆدێل بۆ دۆزینەوە و بەدواداچوونی ئۆتۆمبێلەکان results = model.track(frame, persist=True, classes=[2, 3, 5, 7]) # ئۆتۆمبێل، پاس، لۆری، هتد if results[0].boxes.id is not None: boxes = results[0].boxes.xyxy.cpu().numpy() track_ids = results[0].boxes.id.int().cpu().numpy() confidences = results[0].boxes.conf.cpu().numpy() for box, track_id, conf in zip(boxes, track_ids, confidences): x1, y1, x2, y2 = map(int, box) # کێشانەوەی چوارگۆچە دەوروبەری ئۆتۆمبێلەکە cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2) # نوسینی ID و جۆری ئۆتۆمبێلەکە لەسەر شاشە label = f"ID: {track_id} | {conf:.2f}" cv2.putText(frame, label, (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2) # تۆمارکردنی چرکە بە چرکەی ڤیدۆکە out.write(frame) cap.release() out.release() cv2.destroyAllWindows() print("کارەکە بە سەرکەوتوویی تەواو بوو!")
